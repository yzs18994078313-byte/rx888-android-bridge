cmake_minimum_required(VERSION 3.22)
project(rx888_android_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENABLE_FLOAT ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DISABLE_FORTRAN ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(src/fftw-3.3.10 EXCLUDE_FROM_ALL)
target_include_directories(fftw3f PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src/fftw-3.3.10/api>")
if(ANDROID)
  # FFTW's upstream CMake omits NEON; the source supports AArch64 NEON.
  file(GLOB neon_dft "src/fftw-3.3.10/dft/simd/neon/*.c")
  file(GLOB neon_rdft "src/fftw-3.3.10/rdft/simd/neon/*.c")
  target_sources(fftw3f PRIVATE ${neon_dft} ${neon_rdft})
  target_compile_definitions(fftw3f PRIVATE HAVE_NEON=1)
  set(LIBUSB_SRC "${CMAKE_CURRENT_SOURCE_DIR}/src/libusb-1.0.27" CACHE PATH "" FORCE)
  add_subdirectory(libusb-cmake)
  add_executable(rx888_bridge bridge/server.cpp)
  set_target_properties(rx888_bridge PROPERTIES PREFIX "lib" SUFFIX ".so")
  target_link_libraries(rx888_bridge PRIVATE fftw3f usb1.0 log)
  target_compile_options(rx888_bridge PRIVATE -O3 -Wall -Wextra -Werror=return-type)
  target_link_options(rx888_bridge PRIVATE -Wl,-z,max-page-size=16384)
  add_executable(usb_diagnostics EXCLUDE_FROM_ALL bridge/usb_diagnostics.cpp)
  target_link_libraries(usb_diagnostics PRIVATE usb1.0 log)
  target_link_options(usb_diagnostics PRIVATE -Wl,-z,max-page-size=16384)
  add_executable(stream_check EXCLUDE_FROM_ALL bridge/stream_check.cpp)
  target_link_options(stream_check PRIVATE -Wl,-z,max-page-size=16384)
endif()
add_executable(ddc_tests bridge/ddc_tests.cpp)
target_link_libraries(ddc_tests PRIVATE fftw3f)
add_executable(queue_tests bridge/queue_tests.cpp)
